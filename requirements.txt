from flask import Flask, render_template, request, redirect, url_for, flash, send_from_directory
from flask_mail import Mail, Message
from werkzeug.utils import secure_filename
from datetime import datetime
import os, json, uuid

app = Flask(__name__)
app.secret_key = os.environ.get("SECRET_KEY", "cotizaferre-clave-secreta-2026")

BASE_DIR = os.path.abspath(os.path.dirname(__file__))
UPLOAD_FOLDER = os.path.join(BASE_DIR, "uploads")
DATA_FILE = os.path.join(BASE_DIR, "solicitudes.json")
os.makedirs(UPLOAD_FOLDER, exist_ok=True)

app.config["UPLOAD_FOLDER"] = UPLOAD_FOLDER
app.config["MAX_CONTENT_LENGTH"] = 15 * 1024 * 1024  # 15 MB
ALLOWED_EXTENSIONS = {"pdf", "png", "jpg", "jpeg", "webp", "doc", "docx", "xls", "xlsx"}

# --- CONFIGURACIÓN DE CORREO (FLASK-MAIL) ---
app.config['MAIL_SERVER'] = 'smtp.gmail.com'
app.config['MAIL_PORT'] = 587
app.config['MAIL_USE_TLS'] = True
app.config['MAIL_USERNAME'] = 'daidadalos@gmail.com'
# Obtiene la contraseña segura configurada en Render (o usa la temporal)
app.config['MAIL_PASSWORD'] = os.environ.get('MAIL_PASSWORD', 'clave_temporal_1234')
app.config['MAIL_DEFAULT_SENDER'] = 'daidadalos@gmail.com'

mail = Mail(app)

def allowed_file(filename):
    return '.' in filename and filename.rsplit('.', 1)[1].lower() in ALLOWED_EXTENSIONS

@app.route('/')
def index():
    return render_template('index.html')

@app.route('/solicitar', methods=['POST'])
def solicitar():
    nombre = request.form.get('nombre') or request.form.get('empresa') or 'No especificado'
    telefono = request.form.get('telefono') or request.form.get('whatsapp') or 'No especificado'
    marca = request.form.get('marca') or 'No especificada'
    comentarios = request.form.get('comentarios') or 'Sin comentarios'
    
    # Capturar el archivo adjunto desde el formulario
    file = request.files.get('documento') or request.files.get('archivo')
    
    file_bytes = None
    filename = None
    content_type = None

    if file and file.filename != '' and allowed_file(file.filename):
        filename = secure_filename(file.filename)
        file_bytes = file.read()
        content_type = file.content_type

    # Enviar correo electrónico
    try:
        msg = Message(
            subject=f"📩 Nueva Cotización: {nombre} ({marca})",
            recipients=['daidadalos@gmail.com']
        )
        msg.body = f"""
        ¡Has recibido una nueva solicitud de cotización en CotizaFerre!

        --------------------------------------------------
        📌 DATOS DEL CLIENTE:
        --------------------------------------------------
        • Nombre / Empresa: {nombre}
        • Teléfono / WhatsApp: {telefono}
        • Marca de interés: {marca}
        • Comentarios: {comentarios}
        --------------------------------------------------
        """
        
        # Si se adjuntó archivo, lo pega en el correo
        if file_bytes and filename:
            msg.attach(filename, content_type or 'application/octet-stream', file_bytes)
            
        mail.send(msg)
        flash("Tu solicitud fue enviada correctamente. Nos comunicaremos pronto.", "success")
    except Exception as e:
        print(f"Error enviando correo: {e}")
        flash("Ocurrió un error al enviar el correo, pero intentaremos contactarte.", "error")

    return redirect(url_for('index'))

if __name__ == '__main__':
    app.run(debug=True)
